# 内存泄漏监测工具设计

## 【场景题】
**题目：** 如果要你设计一个内存泄漏监测工具，打开关闭页面时会日志打印内存泄漏的对象，如何实现？

**答案：**

**设计思路：**

```
核心原理：
1. Hook UIViewController的dealloc方法（或viewDidDisappear）
2. 在页面pop/dismiss后延迟检查该VC是否被释放
3. 如果延迟后仍未释放 → 说明有内存泄漏
4. 打印泄漏对象的类名和引用链
```

**实现方案：**

| 方案 | 原理 | 优点 | 缺点 |
|------|------|------|------|
| **Method Swizzling** | 交换dealloc方法 | 简单直接 | 需要runtime |
| **延迟检测** | pop后延迟检查weak指针 | 无侵入 | 有延迟 |
| **Fishhook** | hook C函数 | 可hook系统C函数 | 复杂 |
| ** Milton** | 在dealloc中埋点 | 最简单 | 只能检测VC |

## 【代码示例】
```objective-c
#import <objc/runtime.h>
#import <objc/message.h>

// ============================================
// 方案1：Method Swizzling + 延迟检测（推荐）
// ============================================

@interface MemoryLeakDetector : NSObject
+ (void)startMonitoring;
+ (void)stopMonitoring;
@end

@implementation MemoryLeakDetector

+ (void)startMonitoring {
    static dispatch_once_t onceToken;
    dispatch_once(&onceToken, ^{
        // 交换UIViewController的viewDidDisappear方法
        SEL originalSel = @selector(viewDidDisappear:);
        SEL swizzledSel = @selector(mld_viewDidDisappear:);
        
        Method originalMethod = class_getInstanceMethod([UIViewController class], originalSel);
        Method swizzledMethod = class_getInstanceMethod([UIViewController class], swizzledSel);
        
        method_exchangeImplementations(originalMethod, swizzledMethod);
        
        // 交换 UINavigationController的pop方法
        SEL popOriginalSel = @selector(popViewControllerAnimated:);
        SEL popSwizzledSel = @selector(mld_popViewControllerAnimated:);
        
        Method popOriginal = class_getInstanceMethod([UINavigationController class], popOriginalSel);
        Method popSwizzled = class_getInstanceMethod([UINavigationController class], popSwizzledSel);
        
        method_exchangeImplementations(popOriginal, popSwizzled);
    });
}

+ (void)stopMonitoring {
    // 恢复方法交换（略）
}

@end

// ============================================
// UIViewController的Swizzled方法
// ============================================

@interface UIViewController (MemoryLeakDetect)
@end

@implementation UIViewController (MemoryLeakDetect)

- (void)mld_viewDidDisappear:(BOOL)animated {
    [self mld_viewDidDisappear:animated];  // 调用原方法
    
    // 延迟2秒检测是否被释放
    __weak typeof(self) weakSelf = self;
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2.0 * NSEC_PER_SEC)), 
                   dispatch_get_main_queue(), ^{
        __strong typeof(weakSelf) strongSelf = weakSelf;
        if (strongSelf) {
            // 如果2秒后self还存在，说明可能泄漏了
            NSLog(@"🚨 [MemoryLeak] 检测到泄漏: %@", NSStringFromClass([strongSelf class]));
            NSLog(@"🚨 [MemoryLeak] 对象地址: %p", strongSelf);
            
            // 打印引用链（需要FBRetainCycleDetector等工具辅助）
            [self printRetainCycle:strongSelf];
        }
    });
}

- (void)printRetainCycle:(id)object {
    // 使用FBRetainCycleDetector检测循环引用
    // 或使用runtime遍历对象的属性，手动分析引用关系
    
    Class cls = [object class];
    unsigned int count = 0;
    Ivar *ivars = class_copyIvarList(cls, &count);
    
    NSLog(@"🚨 [MemoryLeak] 对象属性列表:");
    for (int i = 0; i < count; i++) {
        const char *name = ivar_getName(ivars[i]);
        const char *type = ivar_getTypeEncoding(ivars[i]);
        NSLog(@"   - %s : %s", name, type);
    }
    
    free(ivars);
    
    // 提示常见泄漏点
    NSLog(@"🚨 [MemoryLeak] 请检查:");
    NSLog(@"   - delegate是否用了weak?");
    NSLog(@"   - block是否用了__weak?");
    NSLog(@"   - NSTimer是否invalidate?");
    NSLog(@"   - 通知是否removeObserver?");
    NSLog(@"   - KVO是否removeObserver?");
}

@end

// ============================================
// UINavigationController的Swizzled方法
// ============================================

@interface UINavigationController (MemoryLeakDetect)
@end

@implementation UINavigationController (MemoryLeakDetect)

- (UIViewController *)mld_popViewControllerAnimated:(BOOL)animated {
    UIViewController *vc = [self mld_popViewControllerAnimated:animated];
    
    if (vc) {
        // pop的VC，延迟检测
        __weak typeof(vc) weakVC = vc;
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2.0 * NSEC_PER_SEC)), 
                       dispatch_get_main_queue(), ^{
            if (weakVC) {
                NSLog(@"🚨 [MemoryLeak] Pop后泄漏: %@", NSStringFromClass([weakVC class]));
            } else {
                NSLog(@"✅ [MemoryLeak] Pop后正常释放: %@", NSStringFromClass([vc class]));
            }
        });
    }
    
    return vc;
}

@end

// ============================================
// 方案2：简化版 - 只用dealloc日志
// ============================================

@interface BaseViewController : UIViewController
@end

@implementation BaseViewController

- (void)dealloc {
    NSLog(@"✅ %@ 已释放", [self class]);
}

@end

// 使用：所有VC继承BaseViewController
// 如果pop后看不到"已释放"日志，说明泄漏了

// ============================================
// 方案3：使用__weak指针检测
// ============================================

@interface LeakChecker : NSObject
+ (void)checkObject:(id)object afterDelay:(NSTimeInterval)delay name:(NSString *)name;
@end

@implementation LeakChecker

+ (void)checkObject:(id)object afterDelay:(NSTimeInterval)delay name:(NSString *)name {
    __weak id weakObj = object;
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(delay * NSEC_PER_SEC)), 
                   dispatch_get_main_queue(), ^{
        if (weakObj) {
            NSLog(@"🚨 [LeakChecker] %@ 未被释放，可能泄漏", name);
        } else {
            NSLog(@"✅ [LeakChecker] %@ 已正常释放", name);
        }
    });
}

@end

// 使用
- (void)pushDetailVC {
    DetailViewController *detail = [[DetailViewController alloc] init];
    [LeakChecker checkObject:detail afterDelay:2.0 name:@"DetailVC"];
    [self.navigationController pushViewController:detail animated:YES];
}

// ============================================
// 在AppDelegate中启动监测
// ============================================

@implementation AppDelegate

- (BOOL)application:(UIApplication *)application 
    didFinishLaunchingWithOptions:(NSDictionary *)launchOptions {
    
    // 启动内存泄漏监测
    [MemoryLeakDetector startMonitoring];
    
    return YES;
}

@end
```

## 【答题要点】
- 核心原理：Hook UIViewController的生命周期方法，pop/dismiss后延迟检测对象是否释放
- 实现方案：Method Swizzling交换dealloc/viewDidDisappear
- 检测逻辑：用__weak指针，延迟2秒后检查是否为nil
- 如果不为nil → 泄漏，打印类名和属性列表
- 辅助工具：FBRetainCycleDetector可以检测循环引用链
- 提示常见泄漏点：delegate、block、timer、通知、KVO
- 简化方案：BaseViewController的dealloc日志
- 第三方库：MLeaksFinder已经实现了类似功能
- 注意：Swizzling要在dispatch_once中保证只执行一次
- 延迟时间：2秒通常足够让ARC完成释放
