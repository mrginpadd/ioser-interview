## 【案例1：异步网络请求】

**正确代码：**
```objective-c
- (void)viewDidLoad {
    [super viewDidLoad];
    
    // ✅ 使用异步网络请求
    NSURLRequest *request = [NSURLRequest requestWithURL:[NSURL URLWithString:@"https://example.com/api/data"]];
    NSURLSessionDataTask *task = [[NSURLSession sharedSession] dataTaskWithRequest:request completionHandler:^(NSData *data, NSURLResponse *response, NSError *error) {
        if (!error) {
            self.json = [NSJSONSerialization JSONObjectWithData:data options:0 error:nil];
            // 在主线程更新UI
            dispatch_async(dispatch_get_main_queue(), ^{
                [self updateUI];
            });
        }
    }];
    [task resume];
}
```

**优化原因：**
```
优化点：
├── 使用NSURLSession异步请求，不阻塞主线程
├── 网络请求在子线程执行
├── 完成后在主线程更新UI
├── 用户可以正常操作App

效果：
├── 主线程不被阻塞
├── App响应流畅
├── 用户体验良好
```

## 【案例2：for循环中复用对象】

**正确代码：**
```objective-c
- (void)processData {
    NSMutableArray *results = [NSMutableArray array];
    
    // ✅ 复用临时对象
    NSMutableString *tempString = [NSMutableString string];
    for (NSInteger i = 0; i < 10000; i++) {
        [tempString setString:@""];
        [tempString appendFormat:@"Item %ld", (long)i];
        [results addObject:[tempString copy]];
    }
}

// ✅ 另一种优化方式：使用autoreleasepool
- (void)processDataWithAutoreleasePool {
    NSMutableArray *results = [NSMutableArray array];
    
    for (NSInteger i = 0; i < 10000; i++) {
        @autoreleasepool {
            NSString *item = [NSString stringWithFormat:@"Item %ld", (long)i];
            [results addObject:item];
        }
    }
}
```

**优化原因：**
```
优化点（复用对象）：
├── 只创建一个NSMutableString对象
├── 在循环中复用这个对象
├── 减少内存分配次数
├── 降低GC压力

优化点（autoreleasepool）：
├── @autoreleasepool包裹每次迭代
├── 临时对象在每次循环结束时释放
├── 避免内存峰值过高
├── 减少GC次数

效果：
├── 内存峰值降低
├── GC次数减少
├── 避免GC导致的卡顿
```

## 【案例3：TableView使用cell复用】

**正确代码：**
```objective-c
- (UITableViewCell *)tableView:(UITableView *)tableView cellForRowAtIndexPath:(NSIndexPath *)indexPath {
    static NSString *identifier = @"Cell";
    // ✅ 使用cell复用
    UITableViewCell *cell = [tableView dequeueReusableCellWithIdentifier:identifier];
    if (!cell) {
        cell = [[UITableViewCell alloc] initWithStyle:UITableViewCellStyleDefault reuseIdentifier:identifier];
    }
    cell.textLabel.text = self.dataArray[indexPath.row];
    return cell;
}
```

**优化原因：**
```
优化点：
├── 使用dequeueReusableCellWithIdentifier复用cell
├── 只在cell不存在时创建新cell
├── 屏幕上只保留少量cell实例
├── 减少内存分配和视图层级操作

效果：
├── 内存占用稳定
├── 滚动流畅
├── CPU占用率降低
```

## 【案例4：避免离屏渲染】

**正确代码：**
```objective-c
- (void)setupViews {
    UIView *cardView = [[UIView alloc] initWithFrame:CGRectMake(20, 20, 280, 100)];
    
    // ✅ 使用shadowPath避免阴影离屏渲染
    cardView.layer.shadowColor = [UIColor blackColor].CGColor;
    cardView.layer.shadowOpacity = 0.5;
    cardView.layer.shadowRadius = 5;
    cardView.layer.shadowPath = [[UIBezierPath bezierPathWithRect:cardView.bounds] CGPath];
    
    // ✅ 使用图片裁剪实现圆角，避免离屏渲染
    UIImageView *iconView = [[UIImageView alloc] initWithFrame:CGRectMake(10, 10, 80, 80)];
    UIImage *iconImage = [[UIImage imageNamed:@"icon"] imageWithRoundedCorners:CGSizeMake(10, 10)];
    iconView.image = iconImage;
    
    [cardView addSubview:iconView];
    [self.view addSubview:cardView];
}
```

**优化原因：**
```
优化点：
├── shadowPath指定阴影路径，GPU不需要额外计算
├── 图片裁剪在CPU端完成，不触发GPU离屏渲染
├── 避免masksToBounds + cornerRadius的组合

效果：
├── GPU负载降低
├── 帧率稳定在60fps
├── 动画流畅
```

## 【案例5：简化AutoLayout】

**正确代码：**
```objective-c
- (void)setupConstraints {
    // ✅ 使用容器视图减少约束数量
    UIView *containerView = [[UIView alloc] init];
    containerView.translatesAutoresizingMaskIntoConstraints = NO;
    [self.view addSubview:containerView];
    
    // 容器视图的约束
    [NSLayoutConstraint activateConstraints:@[
        [containerView.topAnchor constraintEqualToAnchor:self.view.topAnchor constant:20],
        [containerView.leadingAnchor constraintEqualToAnchor:self.view.leadingAnchor constant:20],
        [containerView.trailingAnchor constraintEqualToAnchor:self.view.trailingAnchor constant:-20],
        [containerView.bottomAnchor constraintEqualToAnchor:self.view.bottomAnchor constant:-20]
    ]];
    
    // 子视图使用Frame布局（性能敏感场景）
    _view1.frame = CGRectMake(0, 0, 50, 50);
    _view2.frame = CGRectMake(60, 0, 50, 50);
    // ...
    
    [containerView addSubview:_view1];
    [containerView addSubview:_view2];
}
```

**优化原因：**
```
优化点：
├── 使用容器视图减少顶层约束数量
├── 性能敏感场景使用Frame替代AutoLayout
├── 减少约束求解的复杂度

效果：
├── CPU计算耗时减少
├── 布局更新更快
├── 滚动流畅
```

## 【案例6：批量更新布局】

**正确代码：**
```objective-c
- (void)animateViews {
    // ✅ 批量更新后统一布局
    [_view setNeedsLayout];
    
    for (NSInteger i = 0; i < 50; i++) {
        _view.frame = CGRectMake(i * 10, 0, 100, 100);
    }
    
    [_view layoutIfNeeded]; // 只计算一次
}
```

**优化原因：**
```
优化点：
├── 使用setNeedsLayout标记需要更新
├── 在循环中只修改属性，不触发计算
├── 循环结束后统一调用layoutIfNeeded
├── 只进行一次布局计算

效果：
├── CPU占用率降低
├── 动画流畅
├── 主线程不被阻塞
```
